<script>

  const BASE = 'https://raw.githubusercontent.com/sitar93/glb/main/';

  let pageName = window.location.pathname.split('/').filter(Boolean).pop() || '';

  pageName = pageName.replace(/\.(html?|php|aspx)$/i, '');



  const mv = document.querySelector('#mv');

  const label = document.querySelector('#progress-label');

  const overlay = document.querySelector('#touch-overlay');

  const hint = document.querySelector('#touch-hint');

  const caseToggleWrap = document.querySelector('#case-toggle-wrap');

  const caseToggle = document.querySelector('#case-toggle');



  function headOk(url) {

    return fetch(url, { method: 'HEAD', mode: 'cors', cache: 'no-cache' })

      .then((r) => {

        if (r.ok) return true;

        if (r.status === 405 || r.status === 501) {

          return fetch(url, {

            method: 'GET',

            mode: 'cors',

            cache: 'no-cache',

            headers: { Range: 'bytes=0-0' },

          }).then((r2) => r2.ok);

        }

        return false;

      })

      .catch(() => false);

  }



  function applyModel(withCase, hasCasePoster) {

    label.style.display = '';

    label.textContent = 'loading 3D model 0%';

    const glb = withCase ? BASE + pageName + '_case.glb' : BASE + pageName + '.glb';

    const poster =

      withCase && hasCasePoster ? BASE + pageName + '_case.png' : BASE + pageName + '.png';

    mv.setAttribute('src', glb);

    mv.setAttribute('poster', poster);

    mv.setAttribute('alt', pageName + (withCase ? '_case_3D' : '_3D'));

  }



  customElements.whenDefined('model-viewer').then(async () => {

    const caseGlbUrl = BASE + pageName + '_case.glb';

    const casePngUrl = BASE + pageName + '_case.png';

    const hasCaseGlb = await headOk(caseGlbUrl);

    const hasCasePoster = hasCaseGlb ? await headOk(casePngUrl) : false;



    if (hasCaseGlb) {

      caseToggleWrap.classList.add('visible');

      caseToggleWrap.setAttribute('aria-hidden', 'false');

      caseToggle.addEventListener('change', () => {

        applyModel(caseToggle.checked, hasCasePoster);

      });

    }



    applyModel(false, hasCasePoster);

  });



  const isMobile = () => window.matchMedia('(pointer: coarse)').matches;



  let inactivityTimer = null;



  function enableOverlay() {

    overlay.style.display = 'block';

    hint.style.display = 'block';

    hint.textContent = isMobile() ? 'pinch to interact' : 'click to interact';

  }



  function disableOverlay() {

    overlay.style.display = 'none';

    hint.style.display = 'none';

    mv.removeAttribute('auto-rotate');

    resetInactivityTimer();

  }



  function resetInactivityTimer() {

    clearTimeout(inactivityTimer);

    inactivityTimer = setTimeout(() => {

      enableOverlay();

    }, 2000);

  }



  mv.addEventListener('load', () => {

    setTimeout(() => {

      label.style.display = 'none';

      enableOverlay();

    }, 300);

  });



  overlay.addEventListener('click', () => {

    if (!isMobile()) disableOverlay();

  });



  overlay.addEventListener('touchstart', (e) => {

    if (e.touches.length >= 2) {

      e.preventDefault();

      disableOverlay();

    }

  }, { passive: false });



  mv.addEventListener('progress', (e) => {

    const p = Math.round(e.detail.totalProgress * 100);

    label.textContent = 'loading 3D model ' + p + '%';

  });



  mv.addEventListener('touchstart', () => {

    mv.removeAttribute('auto-rotate');

    clearTimeout(inactivityTimer);

  }, { passive: true });



  mv.addEventListener('touchmove', (e) => {

    if (e.touches.length === 1) e.stopPropagation();

  }, { passive: true });



  mv.addEventListener('touchend', () => {

    resetInactivityTimer();

  }, { passive: true });



  mv.addEventListener('mouseleave', () => {

    resetInactivityTimer();

  });



  mv.addEventListener('mousemove', (e) => {

    if (e.buttons > 0) clearTimeout(inactivityTimer);

  });



  mv.addEventListener('mouseup', () => {

    resetInactivityTimer();

  });

</script>
