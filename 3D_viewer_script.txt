<script>
  const BASE = 'https://raw.githubusercontent.com/sitar93/glb/main/';

  let pageName = window.location.pathname.split('/').filter(Boolean).pop() || '';
  pageName = pageName.replace(/\.(html?|php|aspx)$/i, '');

  const mv = document.querySelector('#mv');
  const label = document.querySelector('#progress-label');
  const overlay = document.querySelector('#touch-overlay');
  const hint = document.querySelector('#touch-hint');
  const caseModeWrap = document.querySelector('#case-mode-wrap');
  const caseModeButtons = caseModeWrap.querySelectorAll('.case-mode-btn');

  const isMobile = () => window.matchMedia('(pointer: coarse)').matches;

  let inactivityTimer = null;
  let modelEngaged = false;

  let hasLoadedOnce = false;      // Mobile: GLB downloaded at least once
  let hasInteractedOnce = false;  // Desktop: user clicked at least once

  let hasCaseGlb = false;
  let hasCasePoster = false;

  let currentMode = 'standalone';

  let pendingGlb = '';
  let pendingPoster = '';

  function getHintText() {
    if (!isMobile()) return 'click to interact';
    return hasLoadedOnce ? 'pinch to interact' : 'pinch to load 3D model';
  }

  function hideLoadingLabel() {
    label.style.display = 'none';
  }

  function showLoadingLabel(percent) {
    label.style.display = '';
    label.textContent = 'loading 3D model ' + percent + '%';
  }

  function setAutoRotate(on) {
    if (on) {
      mv.setAttribute('auto-rotate', '');
      mv.setAttribute('auto-rotate-delay', '0');
      mv.setAttribute('rotation-per-second', '60deg');
    } else {
      mv.removeAttribute('auto-rotate');
    }
  }

  function updatePendingUrls(withCase) {
    pendingGlb = withCase
      ? BASE + pageName + '_case.glb'
      : BASE + pageName + '.glb';

    pendingPoster =
      withCase && hasCasePoster
        ? BASE + pageName + '_case.png'
        : BASE + pageName + '.png';
  }

  function applyPendingSource() {
    if (!mv.getAttribute('src') || mv.getAttribute('src') !== pendingGlb) {
      mv.setAttribute('src', pendingGlb);
    }
  }

  function headOk(url) {
    return fetch(url, {
      method: 'HEAD',
      mode: 'cors',
      cache: 'no-cache'
    })
      .then((r) => {
        if (r.ok) return true;

        if (r.status === 405 || r.status === 501) {
          return fetch(url, {
            method: 'GET',
            mode: 'cors',
            cache: 'no-cache',
            headers: { Range: 'bytes=0-0' },
          }).then((r2) => r2.ok);
        }

        return false;
      })
      .catch(() => false);
  }

  // Show overlay.
  // Desktop: stop auto-rotate only if the user has already interacted once.
  //          On first load the model should keep spinning while loading.
  // Mobile: always stop auto-rotate (it never starts on mobile).
  function enableOverlay() {
    overlay.style.display = 'block';
    hint.style.display = 'block';
    hint.textContent = getHintText();

    hideLoadingLabel();

    modelEngaged = false;

    if (isMobile() || hasInteractedOnce) {
      setAutoRotate(false);
    }
  }

  // Hide overlay and enable manual interaction.
  // Desktop: stop auto-rotate and start inactivity timer.
  // Mobile: if first load, start GLB download. Otherwise start inactivity timer.
  function disableOverlay() {
    overlay.style.display = 'none';
    hint.style.display = 'none';

    modelEngaged = true;
    hasInteractedOnce = true;

    setAutoRotate(false);

    if (isMobile()) {
      if (!hasLoadedOnce) {
        showLoadingLabel(0);
        applyPendingSource();
      } else {
        resetInactivityTimer();
      }
    } else {
      resetInactivityTimer();
    }
  }

  // Switch between standalone / eurorack.
  // If the model was not spinning (overlay visible or manual interaction active),
  // unlock interaction immediately without requiring another click/pinch.
  // Desktop with active auto-rotate keeps spinning after source change.
  function setMode(mode) {
    const withCase = mode === 'standalone';
    const wasRotating = mv.hasAttribute('auto-rotate');

    currentMode = mode;

    caseModeButtons.forEach((btn) => {
      const active = btn.dataset.mode === mode;

      btn.classList.toggle('active', active);
      btn.setAttribute('aria-pressed', active ? 'true' : 'false');
    });

    updatePendingUrls(withCase);

    mv.setAttribute('poster', pendingPoster);
    mv.setAttribute(
      'alt',
      pageName + (withCase ? '_case_3D' : '_3D')
    );

    // If overlay is visible or model is stopped,
    // unlock interaction immediately
    if (overlay.style.display === 'block' || !wasRotating) {
      overlay.style.display = 'none';
      hint.style.display = 'none';

      clearTimeout(inactivityTimer);

      modelEngaged = true;

      setAutoRotate(false);
    }

    // Mobile without loaded model: do not download anything
    if (isMobile() && !modelEngaged) {
      mv.removeAttribute('src');
      hideLoadingLabel();
      return;
    }

    // Mobile before first load: start download
    if (isMobile() && !hasLoadedOnce) {
      showLoadingLabel(0);
      applyPendingSource();
      return;
    }

    // Desktop or mobile after first load: switch source
    showLoadingLabel(0);
    mv.setAttribute('src', pendingGlb);

    // Desktop with active auto-rotate:
    // resume spinning on the new model
    if (!isMobile() && wasRotating) {
      setAutoRotate(true);
    }
  }

  function resetInactivityTimer() {
    clearTimeout(inactivityTimer);

    inactivityTimer = setTimeout(() => {
      enableOverlay();
    }, 2000);
  }

  customElements.whenDefined('model-viewer').then(async () => {
    hasCaseGlb = await headOk(BASE + pageName + '_case.glb');

    hasCasePoster = hasCaseGlb
      ? await headOk(BASE + pageName + '_case.png')
      : false;

    if (hasCaseGlb) {
      caseModeWrap.classList.add('visible');
      caseModeWrap.setAttribute('aria-hidden', 'false');

      caseModeButtons.forEach((btn) => {
        btn.addEventListener('click', () => {
          if (btn.dataset.mode !== currentMode) {
            setMode(btn.dataset.mode);
          }
        });
      });

      setMode('standalone');

    } else {
      setMode('eurorack');
    }

    // Mobile:
    // no auto-rotate, show "pinch to load 3D model"
    if (isMobile()) {
      setAutoRotate(false);
      enableOverlay();
    }

    // Desktop:
    // auto-rotate starts immediately from HTML attribute,
    // no initial overlay
  });

  mv.addEventListener('load', () => {

    if (isMobile()) {

      if (!modelEngaged) return;

      hasLoadedOnce = true;

      hideLoadingLabel();
      resetInactivityTimer();

      return;
    }

    setTimeout(() => {

      hideLoadingLabel();

      // If the user already interacted,
      // do not show "click to interact" immediately
      if (modelEngaged || hasInteractedOnce) {

        overlay.style.display = 'none';
        hint.style.display = 'none';

        resetInactivityTimer();

        return;
      }

      // First desktop load
      enableOverlay();

    }, 300);
  });

  overlay.addEventListener('click', () => {
    if (!isMobile()) {
      disableOverlay();
    }
  });

  overlay.addEventListener('touchstart', (e) => {
    if (e.touches.length >= 2) {
      e.preventDefault();
      disableOverlay();
    }
  }, { passive: false });

  mv.addEventListener('progress', (e) => {

    const p = Math.round(e.detail.totalProgress * 100);

    if (isMobile()) {

      if (!modelEngaged) return;

      if (p >= 100) {
        hideLoadingLabel();
        return;
      }

      showLoadingLabel(p);

      return;
    }

    showLoadingLabel(p);
  });

  mv.addEventListener('touchstart', () => {
    if (modelEngaged) {
      setAutoRotate(false);
    }

    clearTimeout(inactivityTimer);

  }, { passive: true });

  mv.addEventListener('touchmove', (e) => {
    if (e.touches.length === 1) {
      e.stopPropagation();
    }
  }, { passive: true });

  mv.addEventListener('touchend', () => {
    resetInactivityTimer();
  }, { passive: true });

  mv.addEventListener('mouseleave', () => {
    resetInactivityTimer();
  });

  mv.addEventListener('mousemove', (e) => {
    if (e.buttons > 0) {
      clearTimeout(inactivityTimer);
    }
  });

  mv.addEventListener('mouseup', () => {
    resetInactivityTimer();
  });
</script>
